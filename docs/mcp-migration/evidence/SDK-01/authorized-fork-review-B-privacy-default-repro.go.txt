package main
import("bytes";"fmt";"log/slog";"net/http";"net/http/httptest";"strings";"time";"github.com/modelcontextprotocol/go-sdk/mcp")
func main(){var b bytes.Buffer;l:=slog.New(slog.NewJSONHandler(&b,&slog.HandlerOptions{Level:slog.LevelDebug}));slog.SetDefault(l);s:=mcp.NewServer(&mcp.Implementation{Name:"synthetic",Version:"1"},nil);h:=mcp.NewStreamableHTTPHandler(func(*http.Request)*mcp.Server{return s},&mcp.StreamableHTTPOptions{Stateless:true,JSONResponse:true});r:=httptest.NewRequest("POST","http://localhost/mcp",strings.NewReader(`{"jsonrpc":"2.0","method":"notifications/cancelled","params":{"requestId":"synthetic-id-private","reason":"SYNTHETIC_SECRET_TEXT"}}`));r.Header.Set("Content-Type","application/json");r.Header.Set("Accept","application/json, text/event-stream");w:=httptest.NewRecorder();h.ServeHTTP(w,r);time.Sleep(30*time.Millisecond);fmt.Printf("status=%d logs=%s",w.Code,b.String())}

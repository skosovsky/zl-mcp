package main

import (
 "encoding/json"
 "go/parser"
 "go/token"
 "os"
 "strconv"
)

type record struct { File string `json:"file"`; Imports []string `json:"imports"` }
func main() {
 records:=[]record{}
 for _,name:=range os.Args[1:] {
  if name=="--" {continue}
  file,err:=parser.ParseFile(token.NewFileSet(),name,nil,parser.ImportsOnly)
  if err!=nil {panic(err)}
  row:=record{File:name,Imports:[]string{}}
  for _,spec:=range file.Imports {
   value,err:=strconv.Unquote(spec.Path.Value)
   if err!=nil {panic(err)}
   row.Imports=append(row.Imports,value)
  }
  records=append(records,row)
 }
 if err:=json.NewEncoder(os.Stdout).Encode(records);err!=nil {panic(err)}
}

package probe

import (
 "context"
 "errors"
 "testing"
 "time"

 "github.com/skosovsky/toolsy"
 "github.com/skosovsky/metry"
 "github.com/skosovsky/evaly"
)

type principal struct { Owner bool }
type scope struct { Revision int }

func TestPinnedToolsyScopedProxyExecution(t *testing.T) {
 // Arrange
 calls:=0
 tool,err:=toolsy.NewProxyTool("probe","synthetic probe",[]byte(`{"type":"object","properties":{},"additionalProperties":false}`),func(ctx context.Context, env *toolsy.RunEnv,args []byte,yield func(toolsy.Chunk)error)error{
  calls++
  return yield(toolsy.Chunk{Event:toolsy.EventResult,Data:[]byte(`{"ok":true}`),MimeType:"application/json",TypedResult:map[string]bool{"ok":true}})
 },toolsy.WithOutputSchema(map[string]any{"type":"object","properties":map[string]any{"ok":map[string]any{"type":"boolean"}},"required":[]string{"ok"},"additionalProperties":false}))
 if err!=nil {t.Fatal(err)}
 registry,err:=toolsy.NewRegistry(tool)
 if err!=nil {t.Fatal(err)}
 view,err:=registry.View(toolsy.RegistryViewSpec{ToolNames:[]string{"probe"},Owner:"host",PolicyID:"owner-v1",Policy:toolsy.PolicyFunc(func(ctx context.Context,req toolsy.PolicyRequest)toolsy.Decision{
  typed,e:=toolsy.TypedContext[principal,scope](req.CallContext)
  if e!=nil || !typed.Subject.Owner || typed.Scope.Revision!=1{return toolsy.DenyDecision("denied")}
  return toolsy.AllowDecision()
 })})
 if err!=nil {t.Fatal(err)}
 yielded:=0
 call:=toolsy.ToolCall{ToolName:"probe",Env:toolsy.NewRunEnv(nil),Input:toolsy.ToolInput{ArgsJSON:[]byte(`{}`)},CallContext:toolsy.NewCallContext(principal{true},scope{1})}
 // Act
 err=view.Execute(context.Background(),call,func(c toolsy.Chunk)error{yielded++;return nil})
 // Assert
 if err!=nil || calls!=1 || yielded!=1 {t.Fatalf("execution calls=%d yielded=%d err=%v",calls,yielded,err)}
 // Act
 call.Env=toolsy.NewRunEnv(nil);call.CallContext=toolsy.NewCallContext(principal{false},scope{1})
 err=view.Execute(context.Background(),call,func(c toolsy.Chunk)error{return nil})
 // Assert
 if !errors.Is(err,toolsy.ErrPolicyDenied)||calls!=1 {t.Fatalf("policy leaked dispatch: %v calls%d",err,calls)}
}

func TestPinnedMetryOwnedProviderWrapper(t *testing.T) {
 // Arrange
 ctx,cancel:=context.WithTimeout(context.Background(),time.Second);defer cancel()
 provider,err:=metry.New(ctx,metry.WithServiceName("synthetic-probe"))
 if err!=nil {t.Fatal(err)}
 wrapped:=metry.ExecutorWrap(provider,"probe",func(context.Context,int)(int,error){return 7,nil})
 // Act
 result,err:=wrapped(ctx,1)
 shutdownErr:=provider.Shutdown(ctx)
 // Assert
 if err!=nil||result!=7||shutdownErr!=nil {t.Fatalf("result=%d err=%v shutdown=%v",result,err,shutdownErr)}
 // Compile-check alternative borrowed API; owned mode chosen for host.
 _=metry.FromSDK
}

 // Arrange
 var target evaly.Target[int,int,struct{}]=evaly.TargetFunc[int,int,struct{}](func(context.Context,int,evaly.TrialContext[struct{}])(evaly.TargetResult[int],error){return evaly.TargetResult[int]{},nil})
 config:=evaly.RunConfig[int,int,int,struct{}]{Target:target}
 // Act: incomplete configuration must fail preflight without side effects.
 _,err:=evaly.Run(context.Background(),config)
 // Assert
 if err==nil {t.Fatal("incomplete eval accepted")}
 _=evaly.RunPaired[int,int,int,struct{}]
 _=evaly.Compare
 _=evaly.NewCapture
 var _ evaly.Lifecycle[struct{}]=evaly.LifecycleFuncs[struct{}]{}
}
